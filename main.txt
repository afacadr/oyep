import requests
import time
import random
import base64
import os
import json

# ==============================================================================
# Configuration
# ==============================================================================
GITHUB_API_URL = 'https://api.github.com'
TOKEN_FILE = 'token.txt'
TOTAL_REPOS_TO_CREATE = 2
INSTANCES_PER_REPO = 5

class BColors:
    """Class for terminal colors."""
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKCYAN = '\033[96m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

# ==============================================================================
# Helper Functions
# ==============================================================================

def log(message, color=BColors.ENDC):
    print(f"{color}{message}{BColors.ENDC}")

def random_delay(min_seconds=3, max_seconds=5):
    """Waits for a random duration."""
    delay = random.uniform(min_seconds, max_seconds)
    time.sleep(delay)

def read_token():
    """Reads the PAT from the token file."""
    if not os.path.exists(TOKEN_FILE):
        log(f"[ERROR] '{TOKEN_FILE}' not found. Please create it and paste your PAT inside.", BColors.FAIL)
        return None
    with open(TOKEN_FILE, 'r') as f:
        token = f.read().strip()
    if not token:
        log(f"[ERROR] '{TOKEN_FILE}' is empty. Please paste your PAT inside.", BColors.FAIL)
        return None
    return token

def github_api_request(path, pat, method='GET', data=None):
    """Makes a generic request to the API."""
    headers = {
        'Accept': 'application/vnd.github.v3+json',
        'Authorization': f'token {pat}',
        'Content-Type': 'application/json',
    }
    url = f"{GITHUB_API_URL}{path}"
    
    try:
        response = requests.request(method, url, headers=headers, data=json.dumps(data) if data else None)
        response.raise_for_status()
        if response.status_code == 204 or not response.content:
            return None
        return response.json()
    except requests.exceptions.RequestException as e:
        error_message = f"API request to {method} {url} failed: {e}"
        if e.response is not None:
            try:
                error_details = e.response.json().get('message', e.response.text)
                error_message += f"\nDetails: {error_details}"
            except json.JSONDecodeError:
                error_message += f"\nDetails: {e.response.text}"
        log(f"[ERROR] {error_message}", BColors.FAIL)
        raise

# ==============================================================================
# API Service Functions
# ==============================================================================

def get_user(pat):
    return github_api_request('/user', pat)

def list_user_repos(pat):
    all_repos = []
    page = 1
    while True:
        repos = github_api_request(f'/user/repos?type=owner&per_page=100&page={page}', pat)
        if not repos: break
        all_repos.extend(repos)
        page += 1
    return all_repos

def delete_repo(pat, owner, repo_name):
    return github_api_request(f'/repos/{owner}/{repo_name}', pat, method='DELETE')

def create_repo(pat, repo_name):
    data = {'name': repo_name, 'private': False, 'description': 'Repository for provisioned.', 'auto_init': True}
    return github_api_request('/user/repos', pat, method='POST', data=data)

def create_workflow_file(pat, owner, repo_name, content):
    path = '.github/workflows/vps.yml'
    encoded_content = base64.b64encode(content.encode('utf-8')).decode('utf-8')
    data = {'message': 'Create workflow', 'content': encoded_content}
    return github_api_request(f'/repos/{owner}/{repo_name}/contents/{path}', pat, method='PUT', data=data)

def trigger_workflow(pat, owner, repo_name):
    data = {'ref': 'main'}
    return github_api_request(f'/repos/{owner}/{repo_name}/actions/workflows/vps.yml/dispatches', pat, method='POST', data=data)

# ==============================================================================
# Main Execution Logic
# ==============================================================================
def main():
    """Main function to run the entire process."""
    log(f"{BColors.HEADER}{BColors.BOLD}--- VPS Creator ---{BColors.ENDC}")
    
    pat = read_token()
    if not pat:
        return

    repo_prefix = "vps-project"

    log("Starting process...", BColors.BOLD)

    try:
        # 1. Authenticate once
        random_delay()
        log("Authenticating...", BColors.OKBLUE)
        user = get_user(pat)
        user_login = user['login']
        log(f"Authenticated as {user_login}.", BColors.OKGREEN)

        # 2. Delete all repos once at the beginning
        random_delay()
        log("Fetching list of all repositories for initial cleanup...", BColors.WARNING)
        repos_to_delete = list_user_repos(pat)

        if repos_to_delete:
            log(f"Found {len(repos_to_delete)} repositories to delete. This is a destructive action.", BColors.WARNING)
            for repo in repos_to_delete:
                random_delay()
                log(f"Deleting repository: {repo['full_name']}...", BColors.OKBLUE)
                delete_repo(pat, user_login, repo['name'])
                log(f"Successfully deleted {repo['name']}.", BColors.OKGREEN)
            log("Initial repository cleanup complete.", BColors.OKGREEN)
        else:
            log("No repositories found to delete.", BColors.OKBLUE)

        # 3. Loop to create repos and trigger workflows
        for i in range(1, TOTAL_REPOS_TO_CREATE + 1):
            log(f"\n--- Starting creation for Repository {i} of {TOTAL_REPOS_TO_CREATE} ---", BColors.HEADER)
            
            # Create Repo
            random_delay()
            final_repo_name = f"{repo_prefix.strip()}-{str(int(time.time()))[-6:]}-{i}"
            log(f"[Repo {i}] Creating repository: {final_repo_name}...", BColors.OKBLUE)
            repo_data = create_repo(pat, final_repo_name)
            log(f"[Repo {i}] Successfully created repository: {repo_data['html_url']}", BColors.OKGREEN)
            
            # Create Workflow File
            random_delay()
            log(f"[Repo {i}] Creating workflow file in {user_login}/{final_repo_name}...", BColors.OKBLUE)
            workflow_content = get_workflow_content()
            create_workflow_file(pat, user_login, final_repo_name, workflow_content)
            log(f"[Repo {i}] Workflow file created successfully.", BColors.OKGREEN)
            
            # Trigger Workflows
            random_delay()
            log(f"[Repo {i}] Waiting for Actions to register the new workflow...", BColors.OKBLUE)
            
            for j in range(1, INSTANCES_PER_REPO + 1):
                if j > 1: random_delay(1, 2)
                monitor_message = f" You can monitor runs at {repo_data['html_url']}/actions" if j == 1 else ""
                log(f"[Repo {i}] Triggering workflow ({j}/{INSTANCES_PER_REPO})...{monitor_message}", BColors.OKBLUE)
                trigger_workflow(pat, user_login, final_repo_name)
                
                success_message = f"All {INSTANCES_PER_REPO} workflows for this repo triggered." if j == INSTANCES_PER_REPO else f"Workflow ({j}/{INSTANCES_PER_REPO}) triggered."
                log(f"[Repo {i}] {success_message}", BColors.OKGREEN)
            
            if i < TOTAL_REPOS_TO_CREATE:
                log(f"--- Repository {i} setup complete. Starting next one shortly... ---", BColors.OKCYAN)

        log("\n" + "="*50, BColors.BOLD)
        log("✅ --- Entire process completed successfully! ---", BColors.BOLD + BColors.OKGREEN)
        log("="*50, BColors.BOLD)

    except Exception as e:
        log(f"A critical error stopped the process: {e}", BColors.FAIL)

if __name__ == '__main__':
    main()
