import base64
import random
import string
import requests
import time  # Thêm thư viện time
from nacl import encoding, public

def get_random_string(length):
    """Tạo một chuỗi ngẫu nhiên với độ dài cho trước."""
    letters = string.ascii_lowercase + string.digits
    return "".join(random.choice(letters) for i in range(length))

def create_github_repo(token, repo_name):
    """Tạo một kho lưu trữ mới trên GitHub."""
    print(f"Đang tạo kho lưu trữ '{repo_name}'...")
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    data = {
        "name": repo_name,
        "private": False,
    }
    response = requests.post("https://api.github.com/user/repos", headers=headers, json=data)
    if response.status_code == 201:
        print(f"-> Kho lưu trữ '{repo_name}' đã được tạo thành công.")
        return response.json()
    else:
        print(f"-> Lỗi: Không thể tạo kho lưu trữ. Mã trạng thái: {response.status_code}")
        print(response.json())
        return None

def get_repo_public_key(token, owner, repo_name):
    """Lấy public key của kho lưu trữ để mã hóa secret."""
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    response = requests.get(f"https://api.github.com/repos/{owner}/{repo_name}/actions/secrets/public-key", headers=headers)
    if response.status_code == 200:
        return response.json()
    else:
        print(f"-> Lỗi: Không thể lấy public key của kho lưu trữ. Mã trạng thái: {response.status_code}")
        print(response.json())
        return None

def encrypt_secret(public_key, secret_value):
    """Mã hóa secret bằng public key."""
    public_key = public.PublicKey(public_key.encode("utf-8"), encoding.Base64Encoder())
    sealed_box = public.SealedBox(public_key)
    encrypted = sealed_box.encrypt(secret_value.encode("utf-8"))
    return base64.b64encode(encrypted).decode("utf-8")

def create_repo_secret(token, owner, repo_name, secret_name, encrypted_secret, key_id):
    """Tạo một secret mới trong kho lưu trữ."""
    print(f"Đang tạo secret '{secret_name}'...")
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    data = {
        "encrypted_value": encrypted_secret,
        "key_id": key_id,
    }
    response = requests.put(f"https://api.github.com/repos/{owner}/{repo_name}/actions/secrets/{secret_name}", headers=headers, json=data)
    if response.status_code in [201, 204]:
        print(f"-> Secret '{secret_name}' đã được tạo thành công.")
    else:
        print(f"-> Lỗi: Không thể tạo secret. Mã trạng thái: {response.status_code}")
        print(response.json())

def commit_file_to_repo(token, owner, repo_name, file_path, file_content, commit_message):
    """Commit và đẩy một tệp lên kho lưu trữ."""
    print(f"Đang commit tệp '{file_path}'...")
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    encoded_content = base64.b64encode(file_content.encode("utf-8")).decode("utf-8")
    data = {
        "message": commit_message,
        "content": encoded_content,
    }
    response = requests.put(f"https://api.github.com/repos/{owner}/{repo_name}/contents/{file_path}", headers=headers, json=data)
    if response.status_code == 201:
        print(f"-> Tệp '{file_path}' đã được commit thành công.")
    else:
        print(f"-> Lỗi: Không thể commit tệp. Mã trạng thái: {response.status_code}")
        print(response.json())
